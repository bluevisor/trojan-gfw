cmake_minimum_required(VERSION 3.20)
project(trojan CXX)

set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_SOURCE_DIR}/cmake/")

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Prefer upstream BoostConfig.cmake over the (now-removed in CMake 3.30) FindBoost module.
if(POLICY CMP0167)
    cmake_policy(SET CMP0167 NEW)
endif()

file(GLOB_RECURSE CPP_LIST CONFIGURE_DEPENDS src/*.cpp)

add_executable(trojan ${CPP_LIST})
target_include_directories(trojan PRIVATE src)

if(MSVC)
    target_compile_definitions(trojan PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(trojan PRIVATE -Wall -Wextra)
endif()

option(ENABLE_LTO "Enable link-time optimization for Release builds" OFF)
if(ENABLE_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT _lto_ok OUTPUT _lto_err)
    if(_lto_ok)
        set_property(TARGET trojan PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    else()
        message(WARNING "ENABLE_LTO requested but unsupported by toolchain: ${_lto_err}")
    endif()
endif()

option(ENABLE_NATIVE_ARCH "Build with -march=native (binary is not portable across CPUs)" OFF)
if(ENABLE_NATIVE_ARCH AND NOT MSVC)
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag(-march=native _has_march_native)
    if(_has_march_native)
        target_compile_options(trojan PRIVATE -march=native)
    else()
        message(WARNING "ENABLE_NATIVE_ARCH requested but -march=native is not supported by this compiler")
    endif()
endif()

set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
target_link_libraries(trojan PRIVATE Threads::Threads)

find_package(Boost 1.74.0 REQUIRED COMPONENTS system program_options)
target_link_libraries(trojan PRIVATE Boost::system Boost::program_options)
if(MSVC)
    target_compile_definitions(trojan PRIVATE BOOST_DATE_TIME_NO_LIB)
endif()

find_package(OpenSSL 3.0 REQUIRED)
target_link_libraries(trojan PRIVATE OpenSSL::SSL OpenSSL::Crypto)

option(ENABLE_SSL_KEYLOG "Build with SSL KeyLog support" ON)
if(ENABLE_SSL_KEYLOG)
    target_compile_definitions(trojan PRIVATE ENABLE_SSL_KEYLOG)
endif()

option(ENABLE_TLS13_CIPHERSUITES "Build with TLS1.3 ciphersuites support" ON)
if(ENABLE_TLS13_CIPHERSUITES)
    target_compile_definitions(trojan PRIVATE ENABLE_TLS13_CIPHERSUITES)
endif()

option(ENABLE_MYSQL "Build with MySQL support" ON)
if(ENABLE_MYSQL)
    find_package(MySQL REQUIRED)
    target_include_directories(trojan PRIVATE ${MYSQL_INCLUDE_DIR})
    target_link_libraries(trojan PRIVATE ${MYSQL_LIBRARIES})
    target_compile_definitions(trojan PRIVATE ENABLE_MYSQL)
endif()

option(ENABLE_SOCKET_AUTH "Build with UNIX-socket auth backend (trojan-manager protocol)" OFF)
if(ENABLE_SOCKET_AUTH)
    find_package(Boost 1.75.0 REQUIRED COMPONENTS json)
    target_link_libraries(trojan PRIVATE Boost::json)
    target_compile_definitions(trojan PRIVATE ENABLE_SOCKET_AUTH)
endif()

option(FORCE_TCP_FASTOPEN "Force build with TCP Fast Open support" OFF)
if(FORCE_TCP_FASTOPEN)
    target_compile_definitions(trojan PRIVATE TCP_FASTOPEN=23 TCP_FASTOPEN_CONNECT=30)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL Linux)
    option(ENABLE_NAT "Build with NAT support" ON)
    if(ENABLE_NAT)
        target_compile_definitions(trojan PRIVATE ENABLE_NAT)
    endif()

    option(ENABLE_REUSE_PORT "Build with SO_REUSEPORT support" ON)
    if(ENABLE_REUSE_PORT)
        target_compile_definitions(trojan PRIVATE ENABLE_REUSE_PORT)
    endif()

    option(DISABLE_EPOLL "Disable epoll() support in Boost.Asio (e.g. for proxychains compatibility)" OFF)
    if(DISABLE_EPOLL)
        target_compile_definitions(trojan PRIVATE BOOST_ASIO_DISABLE_EPOLL)
    endif()
endif()

if(APPLE)
    find_library(CoreFoundation CoreFoundation)
    find_library(Security Security)
    target_link_libraries(trojan PRIVATE ${CoreFoundation} ${Security})
endif()

if(WIN32)
    target_link_libraries(trojan PRIVATE wsock32 ws2_32 crypt32)
else()
    set(SYSTEMD_SERVICE AUTO CACHE STRING "Install systemd service")
    set_property(CACHE SYSTEMD_SERVICE PROPERTY STRINGS AUTO ON OFF)
    set(SYSTEMD_SERVICE_PATH /lib/systemd/system CACHE PATH "Systemd service path")
    if(SYSTEMD_SERVICE STREQUAL AUTO)
        if(EXISTS /usr/lib/systemd/system)
            set(SYSTEMD_SERVICE ON)
            set(SYSTEMD_SERVICE_PATH /usr/lib/systemd/system CACHE PATH "Systemd service path" FORCE)
        elseif(EXISTS /lib/systemd/system)
            set(SYSTEMD_SERVICE ON)
            set(SYSTEMD_SERVICE_PATH /lib/systemd/system CACHE PATH "Systemd service path" FORCE)
        endif()
    endif()

    include(GNUInstallDirs)
    install(TARGETS trojan DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES examples/server.json-example DESTINATION ${CMAKE_INSTALL_FULL_SYSCONFDIR}/trojan RENAME config.json)
    set(DEFAULT_CONFIG ${CMAKE_INSTALL_FULL_SYSCONFDIR}/trojan/config.json CACHE STRING "Default config path")
    target_compile_definitions(trojan PRIVATE DEFAULT_CONFIG="${DEFAULT_CONFIG}")
    install(FILES docs/trojan.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
    install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR} FILES_MATCHING PATTERN "*.md")
    install(DIRECTORY examples DESTINATION ${CMAKE_INSTALL_DOCDIR} FILES_MATCHING PATTERN "*.json-example")
    if(SYSTEMD_SERVICE STREQUAL ON)
        set(CONFIG_NAME config)
        configure_file(examples/trojan.service-example trojan.service)
        set(CONFIG_NAME %i)
        configure_file(examples/trojan.service-example trojan@.service)
        install(FILES ${CMAKE_BINARY_DIR}/trojan.service ${CMAKE_BINARY_DIR}/trojan@.service DESTINATION ${SYSTEMD_SERVICE_PATH})
    endif()

    include(CheckSymbolExists)
    check_symbol_exists(daemon "stdlib.h;unistd.h" HAVE_DAEMON)
    if(HAVE_DAEMON)
        target_compile_definitions(trojan PRIVATE HAVE_DAEMON)
    endif()

    enable_testing()
    add_test(NAME LinuxSmokeTest-basic
             COMMAND bash ${CMAKE_SOURCE_DIR}/tests/LinuxSmokeTest/basic.sh ${CMAKE_BINARY_DIR}/trojan)
    add_test(NAME LinuxSmokeTest-fake-client
             COMMAND bash ${CMAKE_SOURCE_DIR}/tests/LinuxSmokeTest/fake-client.sh ${CMAKE_BINARY_DIR}/trojan)
    SET_TESTS_PROPERTIES(LinuxSmokeTest-fake-client PROPERTIES DEPENDS "LinuxSmokeTest-basic")
    if(ENABLE_SOCKET_AUTH)
        set(TROJAN_MANAGER_BIN "" CACHE FILEPATH "Path to trojan-manager binary (for SocketAuth ctest)")
        if(TROJAN_MANAGER_BIN)
            add_test(NAME SocketAuthTest
                     COMMAND bash ${CMAKE_SOURCE_DIR}/tests/SocketAuth/socket-auth.sh ${CMAKE_BINARY_DIR}/trojan ${TROJAN_MANAGER_BIN})
        endif()
    endif()
endif()
